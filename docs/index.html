Wann stand was in den Daten?!

Von 2013-2017
Fahrräder und Elektrofahrräder werden zusammen ausgewiesen als (Elektro-) Fahrrad

Von 2018-2022
Fahrräder und E-Bikes werden separat ausgewiesen Erste E-Scooter werden bei den Fahrrädern hinzugerechnet - oder bei den E-Bikes?

Ab 2023
E-Bikes, E-Scooter und Fahrräder werden getrennt ausgewiesen.

# The data used in this chunk is already loaded and prepared in previous chunks.

# accidents on the mh by time of day and weekday
df_unfaelle_on_mh_by_time <- sf_unfaelle_bike_wien_31287 %>%
    st_filter(sf_mh_buff_13m_wien) %>%
    st_drop_geometry() %>%
    count(UHRZEIT, WOCHENTAG, name = "n") %>%
    mutate(region = "On Mariahilferstr.")

# accidents near the mh by time of day and weekday
df_unfaelle_near_mh_by_time <- sf_unfaelle_bike_wien_31287 %>%
    st_filter(sf_mh_buff_near_wien) %>%
    st_drop_geometry() %>%
    count(UHRZEIT, WOCHENTAG, name = "n") %>%
    mutate(region = "Near Mariahilferstr.")

# accidents in rest of vienna by time of day and weekday
df_unfaelle_vienna_by_time <- sf_unfaelle_bike_wien_31287 %>%
    st_filter(sf_mh_buff_13m_wien, .predicate = st_disjoint) %>%
    st_filter(sf_mh_buff_near_wien, .predicate = st_disjoint) %>%
    st_drop_geometry() %>%
    count(UHRZEIT, WOCHENTAG, name = "n") %>%
    mutate(region = "Rest of Vienna")

# combine data
df_accidents_by_time_weekday_all <- bind_rows(
    df_unfaelle_on_mh_by_time,
    df_unfaelle_near_mh_by_time,
    df_unfaelle_vienna_by_time
)

# Translate weekdays to German and order factors for plotting
vec_time_order <- c(
    "0 bis 6 Uhr",
    "6 bis 9 Uhr",
    "9 bis 12 Uhr",
    "12 bis 15 Uhr",
    "15 bis 18 Uhr",
    "18 bis 24 Uhr"
)
vec_weekday_order_de <- c("Mo", "Di", "Mi", "Do", "Fr", "Sa", "So")

df_accidents_by_time_weekday_all <- df_accidents_by_time_weekday_all %>%
    mutate(
        WOCHENTAG = recode(
            WOCHENTAG,
            "Mon" = "Mo",
            "Tue" = "Di",
            "Wed" = "Mi",
            "Thu" = "Do",
            "Fri" = "Fr",
            "Sat" = "Sa",
            "Sun" = "So"
        ),
        UHRZEIT = factor(UHRZEIT, levels = rev(vec_time_order)),
        WOCHENTAG = factor(WOCHENTAG, levels = vec_weekday_order_de)
    )


# write out the data
p_out_time_of_bike_accidents = here("data_output/time_of_accidents.csv")
df_accidents_by_time_weekday_all %>% 
  pivot_wider(
    names_from = UHRZEIT,
    values_from = n
  ) %>% 
  group_by(region) %>% 
  arrange(WOCHENTAG,.by_group = T) %>% 
  mutate(
    weekday_short = str_sub(WOCHENTAG,1,1),
    .after = WOCHENTAG
  ) %>% 
  select(
    WOCHENTAG, weekday_short, region, # Keep non-time columns at the start
    `0 bis 6 Uhr`, `6 bis 9 Uhr`, `9 bis 12 Uhr`,
    `12 bis 15 Uhr`, `15 bis 18 Uhr`, `18 bis 24 Uhr`
  ) %>%
  # Step 2: Rename the sorted time columns
  rename_with(
    .fn = ~ str_remove(.," Uhr"),
    .cols = contains("bis")
  ) %>% write_csv(p_out_time_of_bike_accidents)
